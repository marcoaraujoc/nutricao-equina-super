-- Query Medicamentos
select e.nome, m.nome, m."formaFarmaceutica", m.unidade, m.apresentacao, m.controlado, m.classificacao, mv.via 
from schs2vet.tb_medicamentos m, schs2vet.tb_medicamento_vias mv, schs2vet.tb_medicamento_especies me, schs2vet.tb_especies e
where m.id = mv."medicamentoId" and
mv."medicamentoId" = me."medicamentoId" and
e.id = me."especieId" and
m.classificacao='Vacina' and
e.nome = 'Equino' 
order by m.nome asc
